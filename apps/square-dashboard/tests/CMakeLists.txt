cmake_minimum_required(VERSION 3.20)
project(capacity_ui_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory(${LVGL_SOURCE} lvgl)
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP LV_MEM_SIZE=4194304 LV_FONT_MONTSERRAT_14=1 LV_FONT_MONTSERRAT_16=1 LV_FONT_MONTSERRAT_18=1 LV_FONT_MONTSERRAT_20=1 LV_FONT_MONTSERRAT_24=1 LV_FONT_MONTSERRAT_32=1)
add_executable(capacity_ui_tests ui_test.cpp)
target_include_directories(capacity_ui_tests PRIVATE .. ${ARDUINOJSON_SOURCE}/src)
target_link_libraries(capacity_ui_tests PRIVATE lvgl)
